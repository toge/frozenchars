cmake_minimum_required(VERSION 3.25)

project(frozenchars_package_consumer LANGUAGES CXX)

if(CONSUMER_KIND STREQUAL "core")
  find_package(frozenchars CONFIG REQUIRED)
  if(TARGET frozenchars::glaze)
    message(FATAL_ERROR "the core consumer loaded the glaze component without requesting it")
  endif()
  add_library(consumer_core OBJECT consumer_core.cpp)
  target_link_libraries(consumer_core PRIVATE frozenchars::frozenchars)
elseif(CONSUMER_KIND STREQUAL "glaze")
  find_package(frozenchars CONFIG REQUIRED COMPONENTS glaze)
  if(NOT frozenchars_glaze_FOUND)
    message(FATAL_ERROR "frozenchars_glaze_FOUND is false after requesting the glaze component")
  endif()
  if(NOT TARGET frozenchars::glaze)
    message(FATAL_ERROR "frozenchars::glaze is missing")
  endif()
  add_executable(consumer_glaze consumer_glaze.cpp)
  target_link_libraries(consumer_glaze PRIVATE frozenchars::glaze)
elseif(CONSUMER_KIND STREQUAL "unknown")
  find_package(frozenchars CONFIG QUIET COMPONENTS unknown)
  if(frozenchars_FOUND)
    message(FATAL_ERROR "an unknown required component was accepted")
  endif()
elseif(CONSUMER_KIND STREQUAL "optional-glaze")
  find_package(frozenchars CONFIG REQUIRED OPTIONAL_COMPONENTS glaze)
  if(frozenchars_glaze_FOUND)
    message(FATAL_ERROR "frozenchars_glaze_FOUND is true without the installed glaze target")
  endif()
  if(TARGET frozenchars::glaze)
    message(FATAL_ERROR "frozenchars::glaze exists without the installed glaze target")
  endif()
elseif(CONSUMER_KIND STREQUAL "missing-glaze")
  find_package(frozenchars CONFIG QUIET COMPONENTS glaze)
  if(frozenchars_FOUND)
    message(FATAL_ERROR "the glaze component was accepted without an installed target")
  endif()
  if(NOT DEFINED frozenchars_glaze_FOUND OR frozenchars_glaze_FOUND)
    message(FATAL_ERROR "frozenchars_glaze_FOUND did not report the missing installed target")
  endif()
else()
  message(FATAL_ERROR "unknown CONSUMER_KIND: ${CONSUMER_KIND}")
endif()
